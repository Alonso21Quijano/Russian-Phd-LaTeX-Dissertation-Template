\begin{frame}[noframenumbering]{Задача проекции визуальных детекций на плоскость движения БТС}
	\begin{minipage}{0.6\linewidth}
	 \fontsize{7}{8}\selectfont
	\textbf{Имея:}
	\begin{itemize}
		\item детекции окружающих транспортных средств (ТС), полученного с камеры БТС в виде обрамляющих рамок с границами параллельными границам изображения: $\{d_0, \dots d_n\}, d_i = \left(u_{min_i}, v_{min_i}, u_{max_i}, v_{max_i}\right) \in \left(\left[0,\dots H-1\right]\times\left[0,\dots, W-1\right]\right)^2$, где $W$, $H$ -- ширина и высота изображения в пикс.,
		\item семантическую сегментацию проезжей части для этого же изображения: $S: \left[0\dots H-1\right]\times\left[0\dots W-1\right] \rightarrow \{0,1\}$,
		\item внутреннюю и внешнюю калибровки камеры,
	\end{itemize}
	\textbf{необходимо оценить} положение окружающих ТС в виде ориентированных ограничивающих рамок на плоскости $OXY$: $\{b_0,\dots, b_n\}, b_i=\left(x_{c,i}, y_{c,i}, \theta_i, w_i, h_i\right)\in\mathbb{R}^2 \times \left[-\pi, \pi\right) \times \mathbb{R}^2$, где $x_{c,i}, y_{c,i}$ -- положение центра рамки на плоскости $OXY$, $\theta_i$ -- угол поворота рамки, $w_i, h_i$ -- ширина и длина рамки.

	Для решения задачи делаются следующие предположения:
	\begin{itemize}
		\item все ТС находятся на проезжей части;
		\item проезжая часть лежит в плоскости $OXY$.
	\end{itemize}
	\end{minipage}
	\begin{minipage}{0.39\linewidth}
		\includegraphics[width=\linewidth]{projection_task}
	\end{minipage}
\end{frame}

\begin{frame}[noframenumbering]{Алгоритм на основе модели L-Shape. Этап 1: проекция}
    \begin{minipage}{0.5\linewidth}
    	\footnotesize
    	\begin{enumerate}
    		\item Берём $N$ равномерно распределённых точек на нижней границе ограничивающей рамки.
    		\item Каждая точка сдвигается к верху изображения пиксель за пикселем, пока не выйдет за пределы проезжей части.
    		\item Строится выпуклая оболочка полученных точек, которая затем проецируется (центральная проекция) на $OXY$: $\{p_0, \dots p_L\}, p_i=(x_i, y_i)$.
    	\end{enumerate}
    \end{minipage}

\begin{frame}[noframenumbering]{Алгоритм на основе модели L-Shape. Этап 2: вписывание в рамку}
	\footnotesize
	\textbf{Дополнительные предположения:} на изображении видны не более двух смежных сторон ТС; смежные стороны ТС ортогональны.

	Оцениваются параметры двух ортогональных прямых $l: y = kx + d_1$ и $l': x = -ky+d_2$, минимизирующие среднеквадратичное отклонение вписываемых точек:

	$$\operatorname*{argmin}_{a}\left(Xa - y \right)^2 \Rightarrow a^* = \left(X^TX\right)^{-1}X^Ty,\quad \text{ESS} = a^{*T}X^TXa - 2 a^{*T}X^Ty - y^Ty$$

	\begin{itemize}
		\item Перебираются все точки разделения $m \in \{1,\dots,N-1\}$ множества вписываемых точек между $l$ и $l'$;
		\item в качестве решения выбирается $a^*$ с наименьшей ESS;
		\item \textbf{предложена реализация} с вычислительной сложностью $\Theta(N)$ за счёт инкрементального пересчёта $X^TX$ и $X^Ty$.
	\end{itemize}
\end{frame}

\begin{frame}[allowframebreaks, noframenumbering]
	\frametitle{Вписывание в рамку: вывод оптимальных параметров}
	\small
	Согласно сделанным предположениям, существует неизвестная заранее точка разделения $m$ вписываемых точек между прямыми $l$ и $l'$:
	\begin{align*}
		\exists m \in \{1,\dots,N - 1\}\hookrightarrow& \forall i \in \{1,\dots,m\}, p_i = (x_i, y_i) \in l;\\
		&\forall i \in \{m+1,\dots,N\},p_i = (x_i, y_i) \in l'
	\end{align*}

	Для некоторого фиксированного $m$, задача минимизации выглядит следующим образом:

	\begin{equation*}
		\operatorname*{argmin}_{k, d_1,d_2}\left(\begin{pmatrix}
			x_1 & 1 & 0 \\
			x_2 & 1 & 0 \\
			\vdots & \vdots  & \vdots  \\
			x_{m} & 1 & 0 \\
			-y_{m + 1} & 0 & 1 \\
			\vdots & \vdots  & \vdots  \\
			-y_{N} & 0 & 1 \\
		\end{pmatrix}
		\begin{pmatrix}
			k \\
			d_1\\
			d_2 \\
		\end{pmatrix} - \begin{pmatrix}
			y_1 \\
			y_2\\
			\vdots \\
			y_{m} \\
			x_{m + 1} \\
			\vdots \\
			x_{N} \\
		\end{pmatrix}\right)^2
	\end{equation*}

	Введя дополнительные обозначения, мы можем переписать это выражение следующим образом:

	\begin{equation*}
		\operatorname*{argmin}_{a}\left(\begin{pmatrix}
			\widehat{x} & 1 & 0 \\
			-\widehat{y} & 0 & 1 \\
		\end{pmatrix}
		a - \begin{pmatrix}
			\widetilde{y} \\
			\widetilde{x} \\
		\end{pmatrix}\right)^2
	\end{equation*}

	И далее:

	\begin{equation*}
		\operatorname*{argmin}_{a}\left(Xa - y \right)^2
	\end{equation*}

	Решением данной задачи минимизации является:

	\begin{align}
		a^* &= \left(X^TX\right)^{-1}X^Ty \label{eq:optimal_lsh_sol}\\
		\text{ESS} &= a^{*T}X^TXa - 2 a^{*T}X^Ty - y^Ty \label{eq:optimal_lsh_sol_mse}
	\end{align}

	Входящие в решение величины выражаются через семь сумм координат вписываемых точек, каждая из которых пересчитывается за фиксированное число операций при переходе от $m$ к $m+1$, что и даёт реализацию перебора со сложностью $\Theta(N)$.
\end{frame}

\begin{frame}[noframenumbering]\frametitle{Метод картографирования с обратной моделью наблюдений}

	Элементы карты  $m_i$ хранят значения:
	$$l_i = l(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k}) = \ln\left(\dfrac{p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})}{1 - p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})}\right),$$
	где $p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})$ -- вероятность нахождения препятствия в пространстве, описываемом $m_i$.

	При таком формате хранимого, получая следующее измерение $\textbf{z}_{s, j}$, значение $m_i$ обновляется по формуле:
	$$l_{i+1} = l_{i} + \underbrace{l(m_i|\textbf{x}_{i+1}, \textbf{z}_{s, j}) - l(m_i)}_{l_+},$$
	где $l(m_i|\textbf{x}_{i+1}, \textbf{z}_{s, j})$ -- \textbf{обратная модель наблюдений}, $l(m_i)$ -- априорное значение $m_i$.

	Значение вероятности восстанавливается по формуле:
	$$p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k}) = \dfrac{\exp{l_i}}{1+\exp{l_i}}$$
\end{frame}

\begin{frame}[noframenumbering]\frametitle{Расширенная карта проходимости пространства}
	\footnotesize
	\begin{itemize}
		\item Сдвиг --- вычислительно дорогая операция для базовой реализации: $O(N)$ операций, где $N$ --- количество элементов карты.
		\item Для снижения вычислительной сложности \textbf{предложено} увеличить размер массива, хранящего карту, до $\widehat{N} > N$.
		\item В общем случае сдвиг требует только модификации координат положения отображаемого участка карты ($O(1)$ операций).
		\item Сдвиг массива требуется только при достижении картой границы массива ($O(\widehat{N})$ операций).
		\item Амортизационная вычислительная сложность операции $O(1)$.
	\end{itemize}

	\centering
	\includegraphics[width=0.95\linewidth]{images/PrefetchedOMF}
\end{frame}

\begin{frame}[noframenumbering]\frametitle{Карта проходимости с разделяемыми блокировками}
	\footnotesize
	\begin{itemize}
		\item Основным препятствием для одновременного добавления на карту нескольких фрагментов является необходимость хранить все элементы карты приведёнными к единому моменту времени.
		\item Чтобы обойти это препятствие, \textbf{предложено} изменить формат элемента карты, добавив к хранимому значению $l_i$ время $t_i$, которое описывает элемент.
		\item Все операции записи/чтения элемента выполняются атомарно. При добавлении фрагмента значение обновляется через сравнение с обменом (Compare and Swap, CAS).
		\item Таким образом, добавление фрагмента может быть выполнено одновременно с другими операциями добавления, если они не приводят к сдвигу массива (сдвиг остаётся блокирующей операцией).
	\end{itemize}

\begin{frame}[noframenumbering]\frametitle{Карта проходимости с обновлением без блокировок}

	\begin{itemize}
		\item Для полного избавления от блокировок \textbf{предложено} разделить массив на два независимых участка.
		\item Участки описывают смежные области пространства.
		\item При приближении к выходу за границы пространства, хранимого в памяти, элементы участка, на котором нет видимой части карты, сбрасываются и начинают описывать область, куда движется видимая часть карты.
