	\begin{itemize}
		\item все ТС находятся на проезжей части;
		\item проезжая часть лежит в плоскости $OXY$.

\begin{frame}[allowframebreaks]
	\frametitle{Алгоритм проекции визуальных детекций на плоскость
движения БТС на основе модели L-Shape}
    \small
    \textbf{Предложен} алгоритм оценки положения ТС на плоскость движения БТС на основе модели L-Shape. Алгоритм состоит из двух шагов: 
    \begin{enumerate}
    	\item проекция нижней границы ТС на плоскость движения БТС,
    	\item вписывание полученной проекции в ориентированную ограничивающую рамку.
    \end{enumerate}
    
    Для предлагаемого алгоритма делается дополнительное предположение:
    \begin{enumerate}
    	\item на изображении видны не более двух смежных сторон ТС,
    	\item смежные стороны ТС ортогональны.
    \end{enumerate}
    
    \framebreak
    \begin{minipage}{0.5\linewidth}
    	\begin{enumerate}
    		\item Берем $N$ равномерно распределенных точек на нижней границе ограничивающей рамки.
    		\item Каждая полученная точка сдвигается к верху изображения пиксель за пикселем, пока она не выйдет за пределы проезжей.
    		\item Строится выпуклая оболочка полученных точек, которая затем проецируется (центральная проекция) на OXY: $\{p_0, \dots p_L\}, p_i=(x_i, y_i)$. 
    	\end{enumerate}
    \end{minipage}
    \begin{minipage}{0.49\linewidth}
	\centering
	\includegraphics[width=\linewidth]{images/l-shape.png}
	\end{minipage}
	
	\framebreak
	Необходимо оценить параметры двух ортогональных прямых: 
	$l: y = kx + d_1$ и $l': x = -ky+d_2$, таких что минимизируется среднеквадратичное отклонение вписываемых точек до соответствующей прямой.
	
	Согласно сделанным ранее предположениям:
	\begin{align*}
		\exists m \in \{1,\dots,N - 1\}\hookrightarrow& \forall i \in \{1,\dots,m\}, p_i = (x_i, y_i) \in l;\\
		&\forall i \in \{m+1,\dots,N\},p_i = (x_i, y_i) \in l'
	\end{align*}
	
	Для некоторого фиксированного $m$, задача минимизации выглядит следующим образом:
	
	\begin{equation*}
		\operatorname*{argmin}_{k, d_1,d_2}\left(\begin{pmatrix}
			x_1 & 1 & 0 \\
			x_2 & 1 & 0 \\
			\vdots & \vdots  & \vdots  \\
			x_{m} & 1 & 0 \\
			-y_{m + 1} & 0 & 1 \\
			\vdots & \vdots  & \vdots  \\
			-y_{N} & 0 & 1 \\
		\end{pmatrix} 
		\begin{pmatrix}
			k \\
			d_1\\
			d_2 \\
		\end{pmatrix} - \begin{pmatrix}
			y_1 \\
			y_2\\
			\vdots \\
			y_{m} \\
			x_{m + 1} \\
			\vdots \\
			x_{N} \\
		\end{pmatrix}\right)^2    
	\end{equation*}
	
	Введя дополнительные обозначения, мы можем переписать это выражение следующим образом:
	
	\begin{equation*}
		\operatorname*{argmin}_{a}\left(\begin{pmatrix}
			\widehat{x} & 1 & 0 \\
			-\widehat{y} & 0 & 1 \\
		\end{pmatrix} 
		a - \begin{pmatrix}
			\widetilde{y} \\
			\widetilde{x} \\
		\end{pmatrix}\right)^2
	\end{equation*}
	
	И далее:
	
	\begin{equation*}
		\operatorname*{argmin}_{a}\left(Xa - y \right)^2
	\end{equation*}
	
	Решением данной задачи минимизации является:
	
	\begin{align}
		a^* &= \left(X^TX\right)^{-1}X^Ty \label{eq:optimal_lsh_sol}\\
		\text{ESS} &= a^{*T}X^TXa - 2 a^{*T}X^Ty - y^Ty \label{eq:optimal_lsh_sol_mse}
	\end{align}
	
	\framebreak
	Итоговый алгоритм вписывания проекции в рамку имеет следующий вид:
	
	\begin{enumerate}
		\item Перебираем все значения $m \in \{1,\dots,N-1\}$
		\item Пересчитываем матрицы $X,y$
		\item Находим $a^*$, MSE для каждой задачи минимизации по формулам~\ref{eq:optimal_lsh_sol},\ref{eq:optimal_lsh_sol_mse}
		\item В качестве финального решения для $l, l'$ выбираем $a^*$ с наименьшим MSE.
		\item Обрамляем вписываемый набор точек линиями параллельными $l$ и $l'$. Область ограниченная полученными линиями -- искомая ограничивающая рамка.
	\end{enumerate}
	
	\textbf{В работе предложена реализация }алгоритма, имеющая вычислительную сложность $O(N)$, где $N$ -- число точек вписываемой проекции.
\end{frame}

\begin{frame}{Алгоритм проекции визуальных детекций на плоскость движения БТС на основе комплексирования данных камеры и лидаров}
    \fontsize{7}{8}\selectfont
	\IncMargin{2em}
	\begin{algorithm}[H]
		\LinesNumbered
		\KwIn{$\{d_1,\dots d_n\}$ -- визуальные детекции, $P$ -- облако точек.}
		\KwOut{$b_i$ -- ориентированные ограничивающие рамки на виде сверху окружающих БТС объектов.}
		Выделить в облаке точек $P$ точки, не принадлежащие поверхности земли: $P_o$\;
		Кластеризовать полученное облако точек $P_o$;
		Спроецировать все кластеры на изображение $I$: $C_i \rightarrow \widehat{C}_i$.\;
		Для каждого спроецированного кластера найти ограничивающую рамку минимального размера: $d_{C_i}$\;
		Используя Венгерский алгоритм найти наилучшее сопоставление $d_i$ с  $d_{C_i}$\;
		В оставшихся кластерах удалить точки, не попадающие при проекции в сопоставленную детекцию $d_i$: $C_i \rightarrow C^-_i$.\;
		\ForEach{$C^-_i$}{
			Используя RANSAC(RANdom SAmple Consensus) найти прямую $l_i$ в $C^-_i$.\;
			Построить ограничивающую рамку $b_i$, со сторонами параллельными $l_i$ и ортогональной ей $l'_i$.\;
		}
		\Return{$\{b_i\}$}
	\end{algorithm}
\end{frame}

\begin{frame}{Метод проекции сегментации проезжей части на изображении на плоскость движения БTC}
	\begin{minipage}{0.6\linewidth}
	\fontsize{6}{7}\selectfont
	\IncMargin{2em}
	\begin{algorithm}[H]
		\LinesNumbered
		\KwIn{$S_I$ -- cемантическая сегментация; $\text{ppm}$ -- разрешение итоговой проекции[пикс/м]; $l$ -- ширина и высота проекции[м]}
		\KwOut{$S_{BEV}$ -- проекция сегментации проезжей части на плоскость движения}
		$s \leftarrow \frac{1}{\text{ppm}}$ \;
		$L \leftarrow l \cdot \text{ppm}$ \;
		$S_{BEV} \leftarrow $ матрица размера $L\times L$\;
		\For{$i \leftarrow 0$ \KwTo $L$}{
			\For{$j \leftarrow 0$ \KwTo $L$}{
				$x \leftarrow -\dfrac{l}{2} + js$\;
				$y \leftarrow -\dfrac{l}{2} + is$\;
				$(u;v) \leftarrow$ проекция $(x,y, \underline{\textbf{z(x,y)}})$ на изображение\;
				\eIf{$(u,v)$ попадает на изображение И $S_I[u][v] ==$ <<проезжая часть>>}{
					$S_{BEV}[u][v] \leftarrow$ <<cвободно>>\;
				}{
					$S_{BEV}[u][v] \leftarrow$ <<неизвестно>>\;
				}
			}
		}
		\Return{$S_{BEV}$}\;
	\end{algorithm}

\begin{frame}{Построение карты проходимости пространства на основе комплексирования разнородных данных}
	\begin{minipage}{0.6\linewidth}

		\textbf{Имея:}
		\begin{itemize}
			\item известную траекторию движения БТС $\textbf{x}_{1:i}, i \in \mathbb{N}, \textbf{x}_i = \langle x_i, y_i, \theta_i \rangle$, где $(x_i, y_i)$ -- координаты БТС, $\theta_i$ -- угол его поворота в глобальной системе координат $O$ в момент времени $t_i$,
			\item измерения, полученные от бортовых датчиков БТС: $\textbf{z}_{1:S,1:k}, S \in \mathbb{N}, k\in\mathbb{N}$ в моменты времени $(t_{s,1}\dots t_{s,k}), s \in \left[1,S\right]$, где $S$ -- число сенсоров,
			\item известные положения датчиков относительно БТС,
		\end{itemize}
		\textbf{необходимо оценить} карту проходимости пространства, ориентированную параллельно осям глобальной системы координат $O$, элементам $m_i$ которой сопоставляется вероятность нахождения препятствий в описываемом ими пространстве.

\begin{frame}{Метод картографирования с обратной моделью наблюдений}
	
	Элементы карты  $m_i$ хранят значения:
	$$l_i = l(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k}) = \ln\left(\dfrac{p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})}{1 - p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})}\right),$$ 
	где $p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k})$ -- вероятность нахождения препятствия в пространстве, описываемом $m_i$.
	 
	При таком формате хранимого, получая следующее измерение $\textbf{z}_{s, j}$, значение $m_i$ обновляется по формуле:
	$$l_{i+1} = l_{i} + \underbrace{l(m_i|\textbf{x}_{i+1}, \textbf{z}_{s, j}) - l(m_i)}_{l_+},$$
	где $l(m_i|\textbf{x}_{i+1}, \textbf{z}_{s, j})$ -- \textbf{обратная модель наблюдений}, $l(m_i)$ -- априорное значение $m_i$.
	
	Значение вероятности восстанавливается по формуле:
	$$p(m_i|\textbf{x}_{1:i}, \textbf{z}_{1:S,1:k}) = \dfrac{\exp{l_i}}{1+\exp{l_i}}$$
\end{frame}

\begin{frame}{Затухание карты проходимости пространства}
	\begin{minipage}{0.65\linewidth}
		\begin{itemize}
			\item В оригинальном алгоритме картографирования с обратной моделью наблюдения делалось предположение о статичности окружающей сцены.
			\item Наличие динамических объектов в сцене может приводит к появлению на карте <<шлейфов>>, которые могут приводить к неэффективным, небезопасным маршрутам БТС или к отказу от дальнейшего движения.
			\item Для исправления такого поведения вводится экспоненциальное затухание значений $m_i$:
			$$l_i = l_{i-1}\exp(-a(t_{i+1} - t_{i})), a > 0$$

\begin{frame}{Принцип построения карты проходимости пространства}
	\footnotesize
	\begin{itemize}
		\item По измерения каждого датчика в отдельном потоке выполнения строится фрагмент карты, хранящий значения $l_+$, которые необходимо добавить на итоговую карту проходимости пространства;

		\item В отдельном потоке выполнения фрагменты извлекаются из очереди и добавляются на карту.
	\end{itemize}

		\item В противном случае сдвиг, поворот и затухание применяются к фрагменту.

\begin{frame}{Расширенная карта проходимости пространства}
	
	\begin{itemize}
	\item Сдвиг -- вычислительно дорогая операция для базовой реализации: $O(N)$ операций, где $N$ количество элементов карты.
	\item Для снижения вычислительной сложности \textbf{предложено} увеличить размер массива, хранящего карту, до $\widehat{N} > N$.
	\item В общем случае сдвиг требует только модификации координат положения отображаемого участка карты ($O(1)$ операций)
	\item Сдвиг массива требуется только при достижении картой границы массива ($O(\widehat{N})$ операций)
	\item Амортизационная вычислительная сложность операции $O(1)$.
	\end{itemize}
	
	\centering

\end{frame}

\begin{frame}{Карта проходимости с разделяемыми блокировками}
	
	\begin{itemize}
		\item Основным препятствием для одновременного добавления на карту нескольких фрагментов, является необходимость хранить все элементы карты приведёнными к единому моменту времени.
		\item Чтобы обойти это препятствие \textbf{предложено} изменить формат элемента карты проходимости пространства, добавив к хранимому значению $l_i$ время  $t_i$, которое описывает элемент.
		\item Все операции записи/чтения элемента выполняются атомарно. При добавлении фрагмента значение обновляется через сравнение с обменом (Compare and Swap, CAS)
		\item Таким образом добавление фрагмента может быть выполнено одновременно с другими операциями добавления, если они не приводят к сдвигу массива (сдвиг остаётся блокирующей операцией).
	\end{itemize}

\end{frame}


\begin{frame}{Карта проходимости с обновлением без блокировок}
	
	\begin{itemize}
		\item Для полного избавления от блокировок \textbf{предложено} разделить массив на два независимых участка. 
		\item Участки описывают смежные области пространства.
		\item При приближении к выходу за границы пространства хранимого в памяти, элементы участка на котором нет видимой части карты сбрасываются и начинают описывать область, куда движется видимая часть карты.
	\end{itemize}
