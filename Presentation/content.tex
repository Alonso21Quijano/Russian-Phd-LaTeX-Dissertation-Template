\begin{frame}{Карта проходимости пространства}
	\footnotesize
	В качестве базовой модели выбрана \textbf{карта проходимости пространства}: пространство разбивается на двумерную сетку $M$, каждой клетке $m_i$ которой сопоставляется вероятность нахождения препятствия $p(m_i)$.

	\begin{minipage}{0.6\linewidth}
		Основания выбора:
		\begin{itemize}
			\item описывает объекты произвольной геометрии, не требуя их классификации;
			\item явно представляет области, о проходимости которых нет информации;
			\item байесовское обновление сводит учёт данных источника к сложению независимо вычисляемых слагаемых $\Rightarrow$ \textbf{принципиально допускает одновременное обновление несколькими источниками}.
		\end{itemize}
	\end{minipage}
	\begin{minipage}{0.39\linewidth}
		\includegraphics[width=\linewidth]{Presentation/images/map_cells}
	\end{minipage}

	$$l_{t,i} = l_{t-1,i} + \underbrace{\ln\dfrac{p(m_i|z_t, x_t)}{1 - p(m_i|z_t, x_t)}}_{\text{обратная модель наблюдения}} - \; l_{0,i}$$
\end{frame}

\begin{frame}{Оценка карты по стоимости кратчайших путей}
	\footnotesize
	\textbf{PFC-MSE} сравнивает эталонную ($GT$) и оцениваемую ($I$) карты по нормированным стоимостям кратчайших путей от БТС до каждой клетки.\footnote{\fontsize{6}{7}\selectfont Horel J.-B. et al. \href{https://convecs.inria.fr/doc/publications/Horel-Baruffa-Rummelhard-et-al-23.pdf}{A Navigation-Based Evaluation Metric for Probabilistic Occupancy Grids}. IEEE ITSC, 2023.}
	\[
		\mathrm{PFC\text{-}MSE}(GT,I) =
		\frac{\sum_c w_c\bigl(GT_{\mathrm{cost}}(c)-I_{\mathrm{cost}}(c)\bigr)^2}{\sum_c w_c},
		\qquad w_c=1-GT(c)\,I(c).
	\]

	Вероятность занятости белых клеток равна 0, серых --- $0{,}2$, чёрных --- 1.

\end{frame}

\begin{frame}{Выводы обзора и требования к разрабатываемым алгоритмам}
	\footnotesize
	Обзор моделей окружающего пространства, датчиков и алгоритмов их обработки позволил сформулировать требования к разрабатываемым модели и алгоритмам её построения:

	\begin{enumerate}
		\item алгоритмы должны обеспечивать \textbf{комплексирование данных камер и лидаров}: ни один из этих типов датчиков в отдельности не даёт одновременно высокого разрешения и непосредственного измерения расстояния \hfill\textit{(гл.~2)}
		\item алгоритмы \textbf{не должны опираться на предположение} о совпадении поверхности движения с плоскостью \hfill\textit{(гл.~2)}
		\item алгоритмы должны выполняться в \textbf{режиме реального времени на бортовом вычислителе} при ограниченных вычислительных ресурсах, что исключает применение методов, требующих инференса крупных нейросетевых моделей для каждого источника данных \hfill\textit{(гл.~2,~4)}
		\item модель должна быть \textbf{масштабируемой} по количеству датчиков и расширяемой под новые их типы \hfill\textit{(гл.~3)}
		\item модель и алгоритм её построения должны допускать \textbf{одновременное обновление несколькими источниками} данных \hfill\textit{(гл.~4)}
	\end{enumerate}
\end{frame}
\note{
    Третье требование задаёт класс алгоритмов, с которыми проводится сравнение в положении 1 и в результатах комплексирования сегментации.
}

\begin{frame}{Выделение точек земли в облаке лидара}
	\fontsize{7}{8}\selectfont
	\IncMargin{2em}
	\begin{algorithm}[H]
		\LinesNumbered
		\KwIn{$P$ --- облако точек лидара.}
		\KwOut{$P_g$ --- точки земли, $P_o$ --- точки препятствий.}
		$P_c \leftarrow P$. Разбить $P_c$ на воксели $V_i$ и отдельно на полярные секторы $S_i$\;
		В каждом $V_i$ найти среднее $m$ и СКО $\sigma$ высоты $z_l$. Удалить точки с $|z_l-m|>\alpha\sigma$\;
		В каждом $S_i$ найти минимум высоты $z_{min}$ и удалить точки с $z_l>z_{min}+T$\;
		Случайно выбрать $N_{\text{train}}$ вокселей в $P_c$. Записать центры масс их точек в $P_{\text{train}}$\;
		По $P_{\text{train}}$ оценить высоту земли $\widehat{z}_i$ в $(x_i,y_i)$ регрессией на гауссовских процессах с ядром Матерна ($\nu=3/2$) для всех $p_i=(x_i,y_i,z_i)\in P$\;
		$P_g \leftarrow \{\}$\;
		\ForEach{$p_i \in P$}{
			Определить сектор $S(p_i)$ точки $p_i$\;
			\lIf{$z_i < \widehat{z}_i + z_{thr}(S(p_i))$}{добавить $p_i$ в $P_g$}
		}
		$P_o \leftarrow P \setminus P_g$\;
		\Return{$P_g, P_o$}\;
	\end{algorithm}

	\smallskip
	\begin{minipage}[t]{0.49\linewidth}
		\centering
		\includegraphics[height=2.55cm]{images/splitting_demo_pc}\par
		\scriptsize Зелёные точки --- земля, красные --- препятствия.
	\end{minipage}\hfill
	\begin{minipage}[t]{0.49\linewidth}
		\centering
		\includegraphics[height=2.55cm]{images/splitting_demo_img}\par
		\scriptsize Та же сцена с бортовой камеры.
	\end{minipage}
\end{frame}
\note{
	Алгоритм~2, с.~62 диссертации. Параметры $\alpha$ и $T$ задают пороги отбора точек для регрессии. Выборка $P_{\text{train}}$ состоит из нескольких сотен центров масс. При классификации исходного облака порог $z_{thr}(S(p_i))$ задаётся для сектора точки $p_i$ с учётом дальности до БТС.
}

\begin{frame}{Алгоритм проекции визуальных детекций на основе комплексирования данных камеры и лидаров}
    \fontsize{7}{8}\selectfont
	\IncMargin{2em}
	\begin{algorithm}[H]
		\LinesNumbered
		\KwIn{$I$ -- изображение с RGB-камеры, $P$ -- облако точек.}
		\KwOut{$b_i$ -- ориентированные ограничивающие рамки на виде сверху окружающих БТС объектов.}
		Произвести поиск двумерных детекций объектов на $I$: $\{d_1,\dots d_n\}$\;
		Выделить в облаке точек $P$ точки, не принадлежащие поверхности земли: $P_o$\;
		Кластеризовать полученное облако точек $P_o$ с расстоянием Евклида\;
		Спроецировать все кластеры на изображение $I$: $C_i \rightarrow \widehat{C}_i$\;
		Для каждого спроецированного кластера найти ограничивающую рамку минимального размера: $d_{C_i}$\;
		Используя венгерский алгоритм найти наилучшее сопоставление $d_i$ с $d_{C_i}$\;
		В оставшихся кластерах удалить точки, не попадающие при проекции в сопоставленную детекцию $d_i$: $C_i \rightarrow C^-_i$\;
		\ForEach{$C^-_i$}{
			Используя RANSAC (RANdom SAmple Consensus) найти прямую $l_i$ в $C^-_i$\;
			Построить ограничивающую рамку $b_i$, со сторонами параллельными $l_i$ и ортогональной ей $l'_i$\;
		}
		\Return{$\{b_i\}$}
	\end{algorithm}
\end{frame}

\begin{frame}{Алгоритм проекции сегментации проезжей части на плоскость движения БТС}
	\begin{minipage}{0.6\linewidth}
	\fontsize{6}{7}\selectfont
	\IncMargin{2em}
	\begin{algorithm}[H]
		\LinesNumbered
		\KwIn{$S_I$ -- семантическая сегментация; $\text{res}$ -- разрешение итоговой проекции [элементов/м]; $l$ -- ширина и высота проекции [м]; $z(x,y)$ -- высота поверхности движения}
		\KwOut{$S_{BEV}$ -- проекция сегментации проезжей части на плоскость движения}
		$s \leftarrow \frac{1}{\text{res}}$ \;
		$L \leftarrow l \cdot \text{res}$ \;
		$S_{BEV} \leftarrow $ матрица размера $L\times L$\;
		\For{$i \leftarrow 0$ \KwTo $L$}{
			\For{$j \leftarrow 0$ \KwTo $L$}{
				$x \leftarrow -\dfrac{l}{2} + js$\;
				$y \leftarrow -\dfrac{l}{2} + is$\;
				$(u;v) \leftarrow$ проекция $(x,y, \underline{\textbf{z(x,y)}})$ на изображение\;
				\eIf{$(u,v)$ попадает на изображение И $S_I[u][v] ==$ <<свободно>>}{
					$S_{BEV}[i][j] \leftarrow$ <<свободно>>\;
				}{
					$S_{BEV}[i][j] \leftarrow$ <<неизвестно>>\;
				}
			}
		}
		\Return{$S_{BEV}$}\;
	\end{algorithm}
	\end{minipage}
	\begin{minipage}{0.39\linewidth}
		\includegraphics[width=\linewidth]{images/NonPlaneProj}
	\end{minipage}
\end{frame}



\begin{frame}{Комплексирование проекции с базовой картой проходимости пространства}
	\fontsize{7}{8.4}\selectfont
	\begin{minipage}[t]{0.57\linewidth}
		Проекция не является самостоятельной моделью: сведений о препятствиях в ней нет. Она объединяется с картой базового лидарного алгоритма:

		\begin{itemize}
			\item элементы <<свободно>> $\rightarrow l^c_+$;
			\item элементы <<занято>> либо <<неизвестно>> в поле зрения камеры $\rightarrow l^c_-$;
			\item остальные элементы $\rightarrow 0$; карты складываются байесовским обновлением.
		\end{itemize}

		Расхождение источников переводит элемент в состояние <<неизвестно>>:
		$$l^c_- + l^b_+ \in (-l_t, l_t)$$

		\textbf{Два типа комплексирования в одном алгоритме:}
		\begin{itemize}
			\item \textbf{кооперативное} при проецировании: лидар --- геометрия поверхности, камера --- семантика;
			\item \textbf{конкурирующее} при объединении карт: оба источника оценивают проходимость элемента.
		\end{itemize}
	\end{minipage}\hfill
	\begin{minipage}[t]{0.4\linewidth}
		\vspace{0pt}\centering
		\includegraphics[width=\linewidth]{images/occlusion_a}

		\fontsize{6}{7}\selectfont Явное состояние <<неизвестно>> сохраняет на карте слепые зоны: их игнорирование при планировании ведёт к поздней реакции на выходящие из них препятствия.
	\end{minipage}
\end{frame}
\note{
    Подробнее о важности учёта слепых зон --- в запасных слайдах.
}
